# MONAI Bundle runtime dependencies.
# MONAI raised from 1.3.x to 1.5.2 for security fixes (bundles that declare
# monai_version 1.3 must be re-validated with this image).
# torch is provided by the base Deep Learning Container.
#
# numpy pinned <2: MONAI 1.3.x is built for NumPy 1.x. Under NumPy 2.x,
# transform seeding hits `OverflowError: 4294967296 out of bounds for uint32`
# (MAX_SEED == 2**32) when instantiating Compose. Pinning avoids it.
# >=1.6.1: pickle/NumpyReader RCE, command injection, torch.load, path traversal,
# zip slip (GHSA-p8cm, 6vm5, x6ww, 9rg3, 89gg, qxq5, rghg, wg9g, 6hp3). Needs torch>=2.8
# (this image's DLC base is torch 2.9).
monai==1.6.1
numpy<2
nibabel>=5.1.0
fire>=0.5.0
# MONAI bundle training loops use PyTorch-Ignite (SupervisedTrainer, handlers,
# ignite.metrics.*). Required to run a bundle's train.json.
pytorch-ignite>=0.4.11
